    Interface PrimitiveContract<TKind, TSlot, TState>

    The shape of a primitive contract — SSOT for generation + validation.

    Generic parameters: TKind - Literal kind identifier ("dialog", "button", "tabs") TSlot - Union of slot names for this primitive TState - Union of state tokens for this primitive

    ariaStateProvider declares WHO provides ARIA+state builders: "meltui" → dsAdapter handles ARIA, 3-folder parity enforced "coldlight" → native HTML handles ARIA, 2-folder parity enforced

    The contract is the COMPLETE specification. It declares every structural element, ARIA requirement, state, keyboard interaction, and behavior that a primitive WILL HAVE when fully built as a component. The contract is adapter-agnostic — it doesn't care if meltui, react-aria, ark/zag, or hand-rolled code fulfills it.

    DESIGN: No unknown types. Every field is typed from a vocabulary. JSON-serializable. Schema-drivable. Powers CSS generation + dev tools.

    interface PrimitiveContract<
        TKind extends string = string,
        TSlot extends string = string,
        TState extends StateToken = StateToken,
    > {
        kind: TKind;
        ariaStateProvider: "meltui" | "coldlight";
        slots: {
            required: readonly TSlot[];
            optional: readonly TSlot[];
            forbidden: readonly string[];
        };
        slotSpec: Record<TSlot, SlotSpec>;
        states: { shared: readonly TState[]; specific: readonly TState[] };
        slotStates: SlotStateMap<TSlot, TState>;
        slotActions: Record<TSlot, readonly SlotAction[]>;
        aria: AriaRequirements;
        keyboard: readonly (
            | "Enter"
            | "Space"
            | "Escape"
            | "Tab"
            | "ArrowUp"
            | "ArrowDown"
            | "ArrowLeft"
            | "ArrowRight"
            | "Home"
            | "End"
            | "PageUp"
            | "PageDown"
            | "Delete"
            | "Backspace"
        )[];
        focusStrategy: "none"
        | "trap"
        | "roving"
        | "activedescendant"
        | "redirect";
        compound: boolean;
        orientation?: "horizontal" | "vertical" | "both";
        dismissBehavior?: DismissBehavior;
    }
