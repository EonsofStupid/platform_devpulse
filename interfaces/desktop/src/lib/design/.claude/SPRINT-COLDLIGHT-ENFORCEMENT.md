## Architecture Summary (LOCKED)

### Single Source of Truth
```
ONE config:         src/lib/design/coldlight.config.ts (orchestrator)
                    + fragments: src/lib/design/config/coldlight/*

ONE build scanner:  src/lib/design/taxonomy/scanner.ts

ONE runtime registry: src/lib/design/taxonomy/registry.ts

ONE generated manifest: src/lib/design/taxonomy/.generated/coldlight.manifest.json
```

### Build-Time vs Runtime (CRITICAL)
- **Build time**: Scanner reads config + folders + contracts → generates `coldlight.manifest.json`
- **Runtime**: Registry reads ONLY the manifest (NO fs scanning, NO taxonomy imports)

### 4-Folder Parity (Enforced)
Every primitive kind requires these 4 folders to exist:
```
taxonomy/primitives/<kind>/           ← Contract SSOT
api/adapters/coldlight/<kind>/        ← ColdLight public API
api/adapters/meltui/_internal/<kind>/ ← MeltUI adapter
primitives/<kind>/                    ← Svelte component
```

### CategoryId vs Kind (Clarified)
- **CategoryId** = Hub grouping (primitives / semantics / composites) → belongs in config
- **Kind** = Folder name inside category (dialog, tabs, button) → belongs in contracts/manifest

```typescript
export type ActionId = string;

  required: readonly S[];
  optional: readonly S[];
  forbidden: readonly S[];
}>;

  shared: readonly T[];
  specific: readonly T[];
}>;

export type SlotStateMap<S extends SlotId, T extends StateToken> = Readonly<

>;

export type Collections<CK extends string> = Readonly<{
  get: (key: CK) => unknown;
}>;

export type PrimitiveContract<
  K extends PrimitiveKind,
  S extends SlotId,
  T extends StateToken,
  CK extends string = never
> = Readonly<{
  kind: K;
  ariaStateProvider: "meltui";
  slots: SlotInventory<S>;
  states: StateInventory<T>;
  slotStates: SlotStateMap<S, T>;
  slotActions: Readonly<Record<S, readonly ActionId[]>>;
  collections: Collections<CK>;
}>;

/** Build-time scan output */
export type ColdLightManifest = Readonly<{
  version: string;
  generatedAt: string;

  adapters: Readonly<{
    enabled: readonly string[];
    defaultActive: string;
    ariaStateProvider: string;
  }>;

  categories: Readonly<{
    primitives: Readonly<{
      kinds: readonly Readonly<{
        kind: string;
        ariaStateProvider: string;
        slots: { required: readonly string[]; optional: readonly string[]; forbidden: readonly string[] };
        states: { shared: readonly string[]; specific: readonly string[] };
        slotStates: Record<string, readonly string[]>;
        slotActions: Record<string, readonly string[]>;
      }>[];
    }>;
  }>;
}>;

export type ScanErrorCode =
  | "PARITY_MISSING_FOLDER"
  | "CONTRACT_LOAD_FAILED"
  | "CONTRACT_INVALID";

export type ScanError = Readonly<{
  code: ScanErrorCode;
  kind: string;
  detail: string;
}>;

export type ScanResult = Readonly<{
  valid: boolean;
```

## PHASE 3: Build-Time Scanner

### Task 3.1: Create scanner.ts
**File**: `src/lib/design/taxonomy/scanner.ts`
**Purpose**: Scan folders, load contracts, generate manifest

```typescript
import type { ColdLightConfig } from "../coldlight.config.js";
import type { PrimitiveContract, ScanError, ScanResult, ColdLightManifest } from "./types.js";

type FsLike = {
  readdirSync: (path: string) => string[];
  statSync: (path: string) => { isDirectory: () => boolean };
};

type ImportLike = (path: string) => Promise<{ default?: unknown; contract?: unknown }>;

const PATHS = {
  taxonomyPrimitives: "taxonomy/primitives",
  apiColdlight: "api/adapters/coldlight",
  apiMeltuiInternal: "api/adapters/meltui/_internal",
  primitives: "primitives",
} as const;

function isDir(fs: FsLike, path: string): boolean {
  try {
    return fs.statSync(path).isDirectory();
  } catch {
    return false;
  }
}

function listKinds(fs: FsLike, baseDesignRoot: string): string[] {
  const root = `${baseDesignRoot}/${PATHS.taxonomyPrimitives}`;
  try {
    return fs
      .readdirSync(root)
      .filter((name) => !name.startsWith("."))
      .filter((name) => isDir(fs, `${root}/${name}`));
  } catch {
    return [];
  }
}

function checkParity(fs: FsLike, baseDesignRoot: string, kind: string): string[] {
  const missing: string[] = [];

  const p1 = `${baseDesignRoot}/${PATHS.taxonomyPrimitives}/${kind}`;
  const p2 = `${baseDesignRoot}/${PATHS.apiColdlight}/${kind}`;
  const p3 = `${baseDesignRoot}/${PATHS.apiMeltuiInternal}/${kind}`;
  const p4 = `${baseDesignRoot}/${PATHS.primitives}/${kind}`;

  if (!isDir(fs, p1)) missing.push(`taxonomy/primitives/${kind}/`);
  if (!isDir(fs, p2)) missing.push(`api/adapters/coldlight/${kind}/`);
  if (!isDir(fs, p3)) missing.push(`api/adapters/meltui/_internal/${kind}/`);
  if (!isDir(fs, p4)) missing.push(`primitives/${kind}/`);

  return missing;
}

async function loadContract(importFn: ImportLike, baseDesignRoot: string, kind: string) {
  const mod = await importFn(`${baseDesignRoot}/${PATHS.taxonomyPrimitives}/${kind}/contract.js`);
  return (mod.default ?? mod.contract) as PrimitiveContract<string, string, string, string> | undefined;
}

function isContractShape(x: unknown): x is PrimitiveContract<string, string, string, string> {
  if (!x || typeof x !== "object") return false;
  const c = x as Record<string, unknown>;
  return (
    typeof c.kind === "string" &&
    c.ariaStateProvider === "meltui" &&
    c.slots !== undefined &&
    c.states !== undefined &&
    c.slotStates !== undefined &&
    c.slotActions !== undefined &&
    c.collections !== undefined &&
    typeof (c.collections as { get?: unknown }).get === "function"
  );
}

export async function scanColdLight(
  fs: FsLike,
  importFn: ImportLike,
  config: ColdLightConfig,
): Promise<ScanResult> {
  const errors: ScanError[] = [];
  const baseDesignRoot = config.roots.designRoot;

  if (!config.generation.categories.primitives.enabled) {
    return { valid: true, errors: [], manifest: null };
  }

  const kinds = listKinds(fs, baseDesignRoot);
  const manifestKinds: ColdLightManifest["categories"]["primitives"]["kinds"] = [];

  for (const kind of kinds) {
    const missingFolders = checkParity(fs, baseDesignRoot, kind);
    for (const m of missingFolders) {
      errors.push({ code: "PARITY_MISSING_FOLDER", kind, detail: m });
    }

    let contract: unknown;
    try {
      contract = await loadContract(importFn, baseDesignRoot, kind);
    } catch {
      errors.push({ code: "CONTRACT_LOAD_FAILED", kind, detail: `taxonomy/primitives/${kind}/contract.ts` });
      continue;
    }

    if (!isContractShape(contract)) {
      errors.push({ code: "CONTRACT_INVALID", kind, detail: `taxonomy/primitives/${kind}/contract.ts` });
      continue;
    }

    manifestKinds.push({
      kind: contract.kind,
      ariaStateProvider: contract.ariaStateProvider,
      slots: {
        required: contract.slots.required,
        optional: contract.slots.optional,
        forbidden: contract.slots.forbidden,
      },
      states: {
        shared: contract.states.shared,
        specific: contract.states.specific,
      },
      slotStates: contract.slotStates as Record<string, readonly string[]>,
      slotActions: contract.slotActions as Record<string, readonly string[]>,
    });
  }

  const valid = errors.length === 0;

  if (!valid && config.enforcement.failOnParity) {
    return { valid: false, errors, manifest: null };
  }

  const manifest: ColdLightManifest = {
    version: "1.0.0",
    generatedAt: new Date().toISOString(),
    adapters: {
      enabled: config.adapters.enabled,
      defaultActive: config.adapters.defaultActive,
      ariaStateProvider: config.adapters.ariaStateProvider,
    },
    categories: {
      primitives: { kinds: manifestKinds },
    },
  };

  if (!valid && config.enforcement.failOnContractLoad) {
    return { valid: false, errors, manifest: null };
  }

  return { valid, errors, manifest };
}
```

### Task 3.2: Create scanner CLI script
**File**: `src/lib/design/scripts/generate-manifest.ts`
**Purpose**: CLI script that runs scanner and writes manifest

```typescript
import * as fs from "node:fs";
import * as path from "node:path";
import { scanColdLight } from "../taxonomy/scanner.js";
import { coldlightConfig } from "../coldlight.config.js";

const DESIGN_ROOT = path.resolve(__dirname, "..");

async function main() {
  console.log("[ColdLight] Scanning primitives...");

  const result = await scanColdLight(
    fs,
    async (p) => import(path.resolve(DESIGN_ROOT, p)),
    coldlightConfig,
  );

  if (!result.valid) {
    console.error("[ColdLight] Scan failed with errors:");
    for (const err of result.errors) {
      console.error(`  [${err.code}] ${err.kind}: ${err.detail}`);
    }
    process.exit(1);
  }

  if (!result.manifest) {
    console.log("[ColdLight] No manifest generated (categories disabled)");
    return;
  }

  const outputPath = path.resolve(DESIGN_ROOT, coldlightConfig.generation.outputs.manifestJsonPath);
  const outputDir = path.dirname(outputPath);

  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  fs.writeFileSync(outputPath, JSON.stringify(result.manifest, null, 2));
  console.log(`[ColdLight] Manifest written to ${outputPath}`);
}

main().catch((err) => {
  console.error("[ColdLight] Fatal error:", err);
  process.exit(1);
});
```

```typescript
import type { ColdLightManifest } from "./types.js";

/**
 * Runtime registry reads ONLY from generated manifest.
 * NO fs scanning. NO taxonomy imports. NO adapter declarations.
 */
import manifestJson from "./.generated/coldlight.manifest.json";

const manifest = manifestJson as ColdLightManifest;

const primitiveIndex = new Map(
  manifest.categories.primitives.kinds.map((k) => [k.kind, k] as const),
);

let activeAdapterId: string = manifest.adapters.defaultActive;

export function getManifest(): ColdLightManifest {
  return manifest;
}

export function getAdapterIds(): readonly string[] {
  return manifest.adapters.enabled;
}

export function getActiveAdapterId(): string {
  return activeAdapterId;
}

export function setActiveAdapterId(next: string): void {
  if (!manifest.adapters.enabled.includes(next)) {
    throw new Error(`[ColdLight] Adapter not enabled: ${next}`);
  }
  activeAdapterId = next;
}

export function getPrimitiveKinds(): readonly string[] {
  return manifest.categories.primitives.kinds.map((k) => k.kind);
}

export function hasPrimitiveKind(kind: string): boolean {
  return primitiveIndex.has(kind);
}

export function getPrimitive(kind: string) {
  const hit = primitiveIndex.get(kind);
  if (!hit) throw new Error(`[ColdLight] Unknown primitive kind: ${kind}`);
  return hit;
}
```

### Task 5.2: Create verify script
**File**: `src/lib/design/scripts/verify-manifest.ts`
**Purpose**: Verify committed manifest matches current scan

```typescript
import * as fs from "node:fs";
import * as path from "node:path";
import { scanColdLight } from "../taxonomy/scanner.js";
import { coldlightConfig } from "../coldlight.config.js";

const DESIGN_ROOT = path.resolve(__dirname, "..");

async function main() {
  console.log("[ColdLight] Verifying manifest...");

  const result = await scanColdLight(
    fs,
    async (p) => import(path.resolve(DESIGN_ROOT, p)),
    coldlightConfig,
  );

  if (!result.valid) {
    console.error("[ColdLight] Verification failed - scan errors:");
    for (const err of result.errors) {
      console.error(`  [${err.code}] ${err.kind}: ${err.detail}`);
    }
    process.exit(1);
  }

  const manifestPath = path.resolve(DESIGN_ROOT, coldlightConfig.generation.outputs.manifestJsonPath);

  if (!fs.existsSync(manifestPath)) {
    console.error("[ColdLight] Manifest file not found. Run 'pnpm coldlight:generate' first.");
    process.exit(1);
  }

  const committed = JSON.parse(fs.readFileSync(manifestPath, "utf-8"));
  const generated = result.manifest;

  // Compare (ignoring generatedAt timestamp)
  const committedNormalized = { ...committed, generatedAt: null };
  const generatedNormalized = { ...generated, generatedAt: null };

  if (JSON.stringify(committedNormalized) !== JSON.stringify(generatedNormalized)) {
    console.error("[ColdLight] Manifest is stale. Run 'pnpm coldlight:generate' and commit.");
    process.exit(1);
  }

  console.log("[ColdLight] Manifest verified - up to date");
}

main().catch((err) => {
  console.error("[ColdLight] Fatal error:", err);
  process.exit(1);
});
```

## Decisions Log

| Decision | Choice | Rationale |
|----------|--------|-----------|
| Config location | `coldlight.config.ts` + fragments | Single orchestrator, modular fragments |
| Manifest format | JSON | Easy to read, parse, commit |
| Scanner runs | Build time only | No runtime fs access needed |
| Registry reads | Manifest only | No taxonomy imports at runtime |
| Adapter support | meltui only (for now) | Only adapter that exists |
| Categories | primitives enabled | semantics/composites future |
